    const TOTAL_QUESTIONS = 10;
    /* ---------- sorteio (shuffle Fisher-Yates) ---------- */
    function shuffle(arr){
    const a = arr.slice();
    for(let i=a.length-1;i>0;i--){
    const j = Math.floor(Math.random()*(i+1));
    [a[i],a[j]]=[a[j],a[i]];
    }
    return a;
    }
    let examQuestions = shuffle(BANK).slice(0, TOTAL_QUESTIONS);
    // embaralha também a ordem das alternativas dentro de cada questão
    examQuestions = examQuestions.map(q => {
    const letters = ["A","B","C","D"];
    const order = shuffle([0,1,2,3]);
    const shuffledOpts = order.map(i => q.opts[i]);
    return {id:q.id, text:q.text, opts:shuffledOpts};
    });

    function renderQuestion(){
    screenQuestion.classList.remove("hidden");
    screenConfirm.classList.add("hidden");
    screenSummary.classList.add("hidden");
    const q = examQuestions[currentIndex];
    document.getElementById("qNumLabel").textContent = `Questão ${currentIndex+1} de ${examQuestions.length}`;
    document.getElementById("qTextLabel").textContent = q.text;
    const wrap = document.getElementById("optionsWrap");

    const letters = ["A","B","C","D"];
    q.opts.forEach((optText, idx) => {

    if(answers[q.id] === optText) label.classList.add("selected");
    const letterSpan = document.createElement("span");
    letterSpan.className = "letter";
    letterSpan.textContent = letters[idx];
    const textSpan = document.createElement("span");
    textSpan.textContent = optText; // evita que tags como <p> sejam interpretadas como HTML
    label.appendChild(letterSpan);
    label.appendChild(textSpan);
    const input = document.createElement("input");
    input.type = "radio";
    input.name = "opt";
    input.value = optText;
    input.style.display = "none";
    if(answers[q.id] === optText) input.checked = true;
    label.prepend(input);
    label.addEventListener("click", () => {

    wrap.querySelectorAll(".opt").forEach(o => o.classList.remove("selected"));

    updateNextButton();
    });
